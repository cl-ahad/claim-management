# =============================================================================
# READY-TO-INSTALL CI WORKFLOW for cl-ahad/claim-management
# -----------------------------------------------------------------------------
# This is the CI pipeline the reporting release needs to be "green" against.
# It is parked here at docs/reporting/ci.yml.txt because the connected GitHub
# token LACKS `workflow` scope, so it cannot be written to .github/workflows/
# directly (the API refuses PUT under .github/workflows/*).
#
# TO INSTALL (one step, once the token has `workflow` scope):
#   move this file to  .github/workflows/ci.yml  (drop the .txt), commit, push.
#   GitHub Actions then runs it on every push to Demo3 and every PR.
#
# Verified correct against the repo on Demo3:
#   - backend/pom.xml       -> Spring Boot 4.1.1, Java 21, tests use H2+Mockito
#                              (@WebMvcTest / MockitoExtension) so NO Postgres
#                              service is required for `mvn verify`.
#   - frontend/package.json -> "build": "tsc -b && vite build"; Vite 7 needs Node 20.
#   - frontend/package-lock.json present -> `npm ci` + npm cache are valid.
# =============================================================================
name: CI

on:
  push:
    branches: [Demo3]
  pull_request:

permissions:
  contents: read

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true

jobs:
  backend:
    name: Backend (Maven verify)
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: backend
    steps:
      - uses: actions/checkout@v4

      - name: Set up JDK 21
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'
          cache: maven

      - name: Build and test
        run: mvn -B --no-transfer-progress clean verify

  frontend:
    name: Frontend (type-check + build)
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: frontend
    steps:
      - uses: actions/checkout@v4

      - name: Set up Node 20
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: npm
          cache-dependency-path: frontend/package-lock.json

      - name: Install dependencies
        run: npm ci

      - name: Type-check and build
        run: npm run build
